{{- /* helpers/autolink for one word of text (no spaces). Returns dict "html" "linked" . Params: word (plain text), plain, root (the site's domain). */ -}} {{- $w := .word -}} {{- $res := slice -}} {{- $linked := false -}} {{- range seq 100 -}} {{- if eq $w "" }}{{ break }}{{ end -}} {{- $u := index (findRESubmatch `(?i)^(.*?)(https?://[^\s<]*|www\.[^\s<]*)` $w 1) 0 | default slice -}} {{- $e := index (findRESubmatch `^(.*?)([a-zA-Z0-9._+-]+@[a-zA-Z0-9_-]+(?:\.[a-zA-Z0-9_-]+)+)` $w 1) 0 | default slice -}} {{- if and (not $u) (not $e) }}{{ $res = $res | append (htmlEscape $w) }}{{ break }}{{ end -}} {{- /* whichever address starts first */ -}} {{- $isURL := and $u (or (not $e) (le (len (index $u 1)) (len (index $e 1)))) -}} {{- $m := cond $isURL $u $e -}} {{- $pre := index $m 1 -}} {{- $addr := index $m 2 -}} {{- $rest := strings.TrimPrefix (printf "%s%s" $pre $addr) $w -}} {{- $ok := true -}} {{- $href := "" -}} {{- if $isURL -}} {{- $www := hasPrefix (lower $addr) "www." -}} {{- /* where an address may start: not straight after a letter or digit (as in "xhttp://" or "awww.") */ -}} {{- $ok = not (findRE `[A-Za-z0-9]$` $pre 1) -}} {{- /* trailing punctuation, an unbalanced ")" and an entity are left out */ -}} {{- range seq 50 -}} {{- $last := substr $addr -1 -}} {{- if in "?!.,:*_~" $last -}} {{- $addr = substr $addr 0 -1 -}} {{- else if and (eq $last ")") (lt (len (findRE `\(` $addr)) (len (findRE `\)` $addr))) -}} {{- $addr = substr $addr 0 -1 -}} {{- else if and (eq $last ";") (findRE `&[A-Za-z0-9]+;$` $addr 1) -}} {{- $addr = replaceRE `&[A-Za-z0-9]+;$` "" $addr -}} {{- else -}} {{- break -}} {{- end -}} {{- end -}} {{- $rest = strings.TrimPrefix (printf "%s%s" $pre $addr) $w -}} {{- /* a valid domain: a dot, no "_" in the last two parts, no empty part */ -}} {{- $domain := index (findRE `^[A-Za-z0-9_.-]*` (replaceRE `(?i)^https?://` "" $addr) 1) 0 | default "" -}} {{- $labels := split $domain "." -}} {{- if or (lt (len $labels) 2) (in $labels "") }}{{ $ok = false }}{{ end -}} {{- if and $ok (ge (len $labels) 2) -}} {{- range last 2 $labels }}{{ if strings.Contains . "_" }}{{ $ok = false }}{{ end }}{{ end -}} {{- end -}} {{- $href = cond $www (printf "http://%s" $addr) $addr -}} {{- else -}} {{- /* an e-mail address can't end with - or _ */ -}} {{- $ok = not (findRE `[-_]$` $addr 1) -}} {{- $href = printf "mailto:%s" $addr -}} {{- end -}} {{- if not $ok -}} {{- /* not an address: keep its first character as text and look further */ -}} {{- $n := add (strings.RuneCount $pre) 1 -}} {{- $res = $res | append (htmlEscape (substr $w 0 $n)) -}} {{- $w = substr $w $n -}} {{- continue -}} {{- end -}} {{- /* percent-encode the address like Markdown renderers do (non-ASCII, \ and the like) */ -}} {{- $enc := slice -}} {{- range split $href "" -}} {{- $enc = $enc | append (cond (findRE `^[A-Za-z0-9;/?:@&=+$,\-_.!~*'()#%]$` . 1) . (urlquery .)) -}} {{- end -}} {{- $href = delimit $enc "" -}} {{- $attrs := "" -}} {{- if and (not $.plain) (hasPrefix (lower $href) "http") (partial "helpers/is-external" (slice $href $.root)) -}} {{- $attrs = ` rel="noopener noreferrer external" target="_blank"` -}} {{- end -}} {{- $res = $res | append (printf `%s%s` (htmlEscape $pre) (htmlEscape $href) $attrs (htmlEscape $addr)) -}} {{- $linked = true -}} {{- $w = $rest -}} {{- end -}} {{- return dict "html" (delimit $res "") "linked" $linked -}}
